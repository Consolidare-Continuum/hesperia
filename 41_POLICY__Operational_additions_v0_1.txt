1. Evidence Quality & Provenance
credibility[s] ∈ [0,1]
recency[s] ∈ [0,1]
consensus[s] ∈ [0,1]
provenanceWeight[s] = credibility[s] * recency[s] * consensus[s]
evidenceQuality(R) = Σ(tau[s] * provenanceWeight[s]) / Σ(tau[s])

2. Multi-Scale Retrieval
embed_side[s]
embed_hex[h]
embed_family[f]
scaleConsistency(R) = cosine(avg(embed_side), avg(embed_hex)) *
                      cosine(avg(embed_hex), avg(embed_family))

3. Temporal Dynamics
timestamp[s]
tau_t[s] = tau[s] * exp(-λ * age[s])
temporalStability(R) = 1 - variance(timestamps of R)

4. Conflict & Counterfactuals
contradictionScore(R) = tensionEnergy(bipolarAxes(A_vec)) * (1 - X_A(R))
counterfactualShift(A_vec, delta[k]) → perturbed A_vec

5. Decision Utility
goalVector[a]
riskProfile[a]
utility(A_vec) = cosine(A_vec, goalVector)
actionScore = utility(A_vec) - Σ(A_vec[a] * riskProfile[a])

6. Causality
causal[a][b]
causalPush[b] = Σ(A_vec[a] * causal[a][b])
causalCoherence = cosine(A_vec, causalPush)

7. Semantic Pressure & Saturation
gradient[s] = avg(|tau[s] - tau[n]|)
pressure(R) = avg(gradient)
saturation(R) = redundancy(embed_side vectors)

8. Retrieval Feedback
retrievalError[f]
retrievalConfidence[f]
adaptiveFamilyWeight[f] = F_vec[f] * retrievalConfidence[f]

9. Domain Calibration

obstacle_threshold_domain[d]
domainFit(d) = -KL(A_vec || domainProfile[d])

10. Summarization Physics
summaryFidelity = cosine(embed(summary), embed(region))
compressionRatio = originalTokens / summaryTokens
IPS = summaryFidelity * (1 - redundancyLoss)

12. Interoperability

13. Composite Metrics
decisionIntegrity(R) =
    evidenceQuality *
    scaleConsistency *
    temporalStability *
    (1 - contradictionScore)

strategicScore =
    actionScore *
    causalCoherence *
    (1 - saturation) *
    evidenceQuality
