HESPERIA — MySQL INTEGRATION & INGESTION v0.1

Scope:
- Describe how HESPERIA uses a MySQL database for:
  - Book/author/chunk/statement storage (source texts).
  - Core semantic schema (`statement`, `side`, etc.).
  - Future attribution and coordinate assignment.
- Provide an auditable log of SQL used and ingestion formats.

Relationships:
- Extends relational schema in: 15_SPEC__Data_and_Schema_v0_1.txt

1. MYSQL COMMANDS (SCHEMA CREATION)
============================================================

The following commands were used to create and evolve the HSPR MySQL
schema. They are listed **without comments**, in executable order.

```sql
CREATE DATABASE IF NOT EXISTS HSPR CHARACTER SET utf8mb4 COLLATE utf8mb4_unicode_ci;

USE HSPR;

CREATE TABLE IF NOT EXISTS authors (
  id BIGINT UNSIGNED NOT NULL AUTO_INCREMENT PRIMARY KEY,
  name VARCHAR(255) NOT NULL UNIQUE,
  created_at TIMESTAMP NOT NULL DEFAULT CURRENT_TIMESTAMP
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_unicode_ci;

CREATE TABLE IF NOT EXISTS books (
  id BIGINT UNSIGNED NOT NULL AUTO_INCREMENT PRIMARY KEY,
  title VARCHAR(512) NOT NULL,
  author_id BIGINT UNSIGNED NOT NULL,
  code VARCHAR(64) NULL UNIQUE,
  chunks_json_path VARCHAR(1024) NOT NULL,
  statements_json_path VARCHAR(1024) NULL,
  meta JSON NULL,
  created_at TIMESTAMP NOT NULL DEFAULT CURRENT_TIMESTAMP,
  CONSTRAINT fk_books_author FOREIGN KEY (author_id) REFERENCES authors(id)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_unicode_ci;

CREATE TABLE IF NOT EXISTS chunks (
  id BIGINT UNSIGNED NOT NULL AUTO_INCREMENT PRIMARY KEY,
  book_id BIGINT UNSIGNED NOT NULL,
  json_id INT NOT NULL,
  text LONGTEXT NOT NULL,
  meta JSON NULL,
  source_file_path VARCHAR(1024) NOT NULL,
  created_at TIMESTAMP NOT NULL DEFAULT CURRENT_TIMESTAMP,
  CONSTRAINT fk_chunks_book FOREIGN KEY (book_id) REFERENCES books(id),
  UNIQUE KEY uq_chunks_book_json_id (book_id, json_id)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_unicode_ci;

CREATE TABLE IF NOT EXISTS statements (
  id BIGINT UNSIGNED NOT NULL AUTO_INCREMENT PRIMARY KEY,
  book_id BIGINT UNSIGNED NOT NULL,
  chunk_json_id INT NULL,
  json_id INT NOT NULL,
  semantic_statement_id BIGINT UNSIGNED NULL,
  text LONGTEXT NOT NULL,
  meta JSON NULL,
  source_file_path VARCHAR(1024) NOT NULL,
  created_at TIMESTAMP NOT NULL DEFAULT CURRENT_TIMESTAMP,
  CONSTRAINT fk_statements_book FOREIGN KEY (book_id) REFERENCES books(id),
  UNIQUE KEY uq_statements_book_json_id (book_id, json_id)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_unicode_ci;

CREATE TABLE IF NOT EXISTS area (
  id SMALLINT NOT NULL PRIMARY KEY,
  name VARCHAR(32) NOT NULL,
  ring_index SMALLINT NOT NULL
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_unicode_ci;

CREATE TABLE IF NOT EXISTS family (
  id SMALLINT NOT NULL PRIMARY KEY,
  area_id SMALLINT NOT NULL,
  local_index SMALLINT NOT NULL,
  name VARCHAR(64) NOT NULL,
  UNIQUE KEY uq_family_area_local (area_id, local_index),
  CONSTRAINT fk_family_area FOREIGN KEY (area_id) REFERENCES area(id)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_unicode_ci;

CREATE TABLE IF NOT EXISTS hex (
  id BIGINT NOT NULL PRIMARY KEY,
  family_id SMALLINT NOT NULL,
  hex_index SMALLINT NOT NULL,
  UNIQUE KEY uq_hex_family_index (family_id, hex_index),
  CONSTRAINT fk_hex_family FOREIGN KEY (family_id) REFERENCES family(id)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_unicode_ci;

CREATE TABLE IF NOT EXISTS statement (
  id BIGINT UNSIGNED NOT NULL AUTO_INCREMENT PRIMARY KEY,
  raw_text LONGTEXT NOT NULL,
  area_id SMALLINT NOT NULL,
  family_id SMALLINT NOT NULL,
  created_at TIMESTAMP NOT NULL DEFAULT CURRENT_TIMESTAMP,
  inward_vector BLOB NULL,
  outward_vector BLOB NULL,
  f_vector BLOB NULL,
  a_vector BLOB NULL,
  CONSTRAINT fk_statement_area FOREIGN KEY (area_id) REFERENCES area(id),
  CONSTRAINT fk_statement_family FOREIGN KEY (family_id) REFERENCES family(id)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_unicode_ci;

CREATE TABLE IF NOT EXISTS side (
  side_id INT NOT NULL PRIMARY KEY,
  area_id SMALLINT NOT NULL,
  family_id SMALLINT NOT NULL,
  hex_index SMALLINT NOT NULL,
  side_index SMALLINT NOT NULL,
  tau DOUBLE NOT NULL DEFAULT 0.0,
  statement_id BIGINT UNSIGNED NULL,
  last_updated TIMESTAMP NOT NULL DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP,
  CONSTRAINT fk_side_area FOREIGN KEY (area_id) REFERENCES area(id),
  CONSTRAINT fk_side_family FOREIGN KEY (family_id) REFERENCES family(id),
  CONSTRAINT fk_side_statement FOREIGN KEY (statement_id) REFERENCES statement(id)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_unicode_ci;

CREATE INDEX side_family_area_idx ON side(family_id, area_id);
CREATE INDEX side_tau_idx ON side(tau DESC);

CREATE TABLE IF NOT EXISTS dictionary_word (
  lemma VARCHAR(128) NOT NULL PRIMARY KEY,
  family_id SMALLINT NOT NULL,
  area_id SMALLINT NOT NULL,
  source VARCHAR(64) NOT NULL,
  updated_at TIMESTAMP NOT NULL DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP,
  CONSTRAINT fk_dict_family FOREIGN KEY (family_id) REFERENCES family(id),
  CONSTRAINT fk_dict_area FOREIGN KEY (area_id) REFERENCES area(id)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_unicode_ci;

CREATE TABLE IF NOT EXISTS side_provenance (
  side_id INT NOT NULL PRIMARY KEY,
  credibility DOUBLE NOT NULL DEFAULT 1.0,
  recency DOUBLE NOT NULL DEFAULT 1.0,
  consensus DOUBLE NOT NULL DEFAULT 1.0,
  CONSTRAINT fk_side_prov_side FOREIGN KEY (side_id) REFERENCES side(side_id)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_unicode_ci;

CREATE TABLE IF NOT EXISTS domain_profile (
  id INT NOT NULL AUTO_INCREMENT PRIMARY KEY,
  name VARCHAR(64) NOT NULL UNIQUE,
  area_vec BLOB NOT NULL
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_unicode_ci;

ALTER TABLE statements
  ADD CONSTRAINT fk_statements_semantic
  FOREIGN KEY (semantic_statement_id) REFERENCES statement(id);
```

============================================================
2. DATABASE STRUCTURE (CONCEPTUAL)
============================================================

Layers:
- **Source text layer** — books, chunks, and source statements.
- **Semantic layer** — HESPERIA statements and sides (coordinates).
- **Dictionary / provenance** — ontology evolution and evidence.

Source text tables:
- `authors` — unique authors; referenced by `books`.
- `books` — each logical work; stores JSON paths and optional metadata:
  - `code` (e.g. `2_TMS-AS`), `chunks_json_path`, `statements_json_path`, `meta`.
- `chunks` — raw book text segmented into chunks:
  - `json_id` mirrors chunk IDs from JSON.
  - `text` holds the chunk content, `meta` holds future extra fields.
  - `source_file_path` keeps the relative path back to the JSON.
- `statements` — (future) book‑source statements:
  - `json_id` mirrors statement ID from JSON.
  - `chunk_json_id` optionally links to `chunks.json_id`.
  - `semantic_statement_id` links forward into the semantic `statement` table.
  - `meta` stores future per‑statement metadata.

Semantic layer:
- `area`, `family`, `hex` — geometry and indexing (per 13_SPEC).
- `statement` — semantic statements ready for retrieval:
  - `raw_text`, `area_id`, `family_id`, `inward_vector`, `outward_vector`,
    `f_vector`, `a_vector`.
- `side` — full coordinate and placement:
  - `side_id`, `area_id`, `family_id`, `hex_index`, `side_index`, `tau`,
    `statement_id`, `last_updated`.

Dictionary and provenance:
- `dictionary_word` — lemma → (family, area, source, updated_at).
- `side_provenance` — credibility metrics per side.
- `domain_profile` — domain‑level area vectors.

Bridging:
- `statements.semantic_statement_id` points from book/source statements to
  their semantic counterpart in `statement.id`.
- `side.statement_id` then connects semantic statements to coordinates.

============================================================
4. JSON INGESTION FORMAT
============================================================

4.1 Chunk JSON (current)
------------------------

Path:
- `text_chunks/HSPR_CHUNKED_2_TMS_AS.json`

Structure:

```json
{
  "book": "The Theory of Moral Sentiments",
  "author": "Adam Smith",
  "chunks": [
    {
      "id": 1,
      "text": "The Theory of Moral Sentiments\\nAmartya Sen\\n..."
    },
    {
      "id": 2,
      "text": "PART I. - OF THE PROPRIETY OF ACTION.\\n\\n"
    }
  ]
}
```

Semantics:
- Top‑level fields:
  - `book` — human‑readable title.
  - `author` — human‑readable author.
  - `chunks[]` — ordered list of chunks.
- Each `chunks[i]`:
  - `id` — integer chunk id (1‑based here), stored as `chunks.json_id`.
  - `text` — raw chunk text, stored as `chunks.text`.
  - Any additional keys will be mirrored into `chunks.meta` as JSON.

4.2 Statement JSON (future)
---------------------------

Reserved path:
- `statements/2_TMS-AS.json`

Intended structure (to be finalized):

```json
{
  "book_code": "2_TMS-AS",
  "statements": [
    {
      "id": 1,
      "chunk_id": 10,
      "text": "A candidate statement extracted from chunks...",
      "area_hint": 3,
      "family_hint": 42
    }
  ]
}
```

Mapping (planned):
- `book_code` → `books.code`.
- `statements[].id` → `statements.json_id`.
- `statements[].chunk_id` → `statements.chunk_json_id`.
- `statements[].text` → `statements.text`.
- Additional keys (`area_hint`, `family_hint`, etc.) → `statements.meta`.

============================================================
5. INGESTION FLOW (SOURCE LAYER)
============================================================

5.1 Loader behavior (mysql_loader.py)
-------------------------------------

Configuration:
- Reads MySQL config from:
  - Environment: `MYSQL_HOST`, `MYSQL_PORT`, `MYSQL_USER`,
    `MYSQL_PASSWORD`, `MYSQL_DB`.
  - Fallback: `.env` in repo root, parsing lines that start with `MYSQL_`.

Author and book:
- Ensures a row in `authors` for the JSON `author` name.
- Ensures a row in `books` for the JSON `book` title and author:
  - `code` is set to `2_TMS-AS` for this book.
  - `chunks_json_path` and `statements_json_path` recorded.

Chunks:
- Parses `chunks[]` from the JSON.
- For each chunk:
  - Writes `json_id`, `text`, and `source_file_path`.
  - Collects any extra keys into `chunks.meta` as a JSON object.
- Uses `ON DUPLICATE KEY` to update existing rows.
- Before loading, it checks:
  - `SELECT COUNT(*) FROM chunks WHERE book_id = :bookId`
  - If count ≥ `len(chunks)` and `len(chunks) > 0`, it skips re‑ingestion.

5.2 Idempotence guarantees
--------------------------

Rerunning `python3 -m sphere_rag.mysql_loader`:
- Does not create duplicate authors or books.
- Does not create duplicate chunks:
  - `UNIQUE (book_id, json_id)` in `chunks`.
  - Upserts on that key.
- Skips chunk re‑ingestion entirely when counts match, for efficiency.

6. ATTRIBUTION AND COORDINATES (SEMANTIC LAYER)
============================================================

6.1 High-level pipeline
-----------------------

Given a chosen source statement from the `statements` table:
1. **Attribution**:
   - HESPERIA’s semantic engine processes the text and computes:
     - `area_id`, `family_id` (dominant area/family).
     - `F_statement`, `A_statement`.
     - `inward_vector[12]`, `outward_vector[12]`.
2. **Semantic statement insertion**:
   - Insert a row into `statement` with:
     - `raw_text`, `area_id`, `family_id`,
       `inward_vector`, `outward_vector`, `f_vector`, `a_vector`.
3. **Link back to source**:
   - Update `statements.semantic_statement_id` to point at the new
     `statement.id`.
4. **Coordinate assignment**:
   - Using the indexing formulas (13_SPEC) and GeometryService:
     - Compute `side_id`, `hex_index`, `side_index` from area/family/hex.
   - Upsert a row into `side`:
     - `side_id`, `area_id`, `family_id`, `hex_index`, `side_index`,
       `tau`, `statement_id`.

6.2 Storage contracts
---------------------

Contract between source and semantic layers:
- Source:
  - `statements.id` is the primary key for book‑source statements.
  - `statements.semantic_statement_id` is NULL until attribution.
- Semantic:
  - `statement.id` is the primary key for attributed statements.
  - `side.statement_id` points to `statement.id`.

Once a `statements` row has a non‑NULL `semantic_statement_id` and the
corresponding `statement` row has at least one `side` entry, the system
can treat this source statement as fully “placed” on the sphere.

7. ATTRIBUTION HELPER (PYTHON)
============================================================

7.1 Purpose
-----------

File:
- `sphere_rag/attribution_helper.py`

Goal:
- Provide a small, DB‑centric utility to:
  - Create a semantic `statement` row from a source statement.
  - Update `statements.semantic_statement_id`.
  - Optionally upsert a `side` row for that semantic statement.

Responsibilities:
- Does **not** perform attribution math itself:
  - Vectors and coordinates must already be computed by the semantic
    engine and geometry/indexing services.
- Encapsulates the exact SQL patterns from 15_SPEC:
  - `INSERT INTO statement(...) VALUES(...)`
  - `INSERT INTO side(...) ... ON DUPLICATE KEY UPDATE ...`

7.2 Usage sketch
----------------

Conceptual recipe:
1. Run attribution on a source statement (text from `statements.text`).
2. Compute area/family and vectors.
3. Compute `side_id`, `hex_index`, `side_index` from coordinates.
4. Call the helper to:
   - Insert into `statement`.
   - Link `statements.semantic_statement_id`.
   - Upsert `side` with tau and the new `statement_id`.

This keeps the DB structure ready for future attribution work while
allowing the coordinate and vector logic to evolve independently.

8. EXAMPLE ATTRIBUTION FLOW (PYTHON)
============================================================

8.1 Example module
------------------

File:
- `sphere_rag/example_attribution_flow.py`

Purpose:
- Demonstrate an end‑to‑end flow **before** the real attribution engine
  is plugged in:
  - Select a chunk from `chunks`.
  - Create or reuse a `statements` row backed by that chunk.
  - Run a dummy attribution function.
  - Insert a semantic `statement` row and assign a `side`.

Key components:
- `DummyAttributionResult` — lightweight container for:
  - `area_id`, `family_id`
  - placeholder vectors (`inward_vector`, `outward_vector`, `f_vector`, `a_vector`)
  - `SideCoord` and `tau`.
- `_dummy_attribution(text: str) -> DummyAttributionResult`
  - Currently pins everything to `area_id = 0`, `family_id = 0`.
  - Derives `hex_index`, `side_index`, and `side_id` from text length.
  - Leaves vectors as `None` (real engine will provide serialized arrays).
- `_ensure_area_family(conn, area_id, family_id)`
  - Inserts minimal placeholder rows into `area` and `family` if missing.
  - Ensures foreign keys for `statement.area_id` and `statement.family_id`
    are satisfied in this example.
- `_ensure_source_statement_from_chunk(conn, book_code, chunk_json_id)`
  - Looks up a `chunks` row for the given book and `json_id`.
  - Inserts a corresponding `statements` row if none exists yet.
  - Returns `statements.id`.
- `run_example_for_first_chunk(...) -> int`
  - Reads text from `chunks` for `(book_code="2_TMS-AS", json_id=1)` by default.
  - Ensures a source `statements` row exists.
  - Runs `_dummy_attribution`.
  - Calls `insert_semantic_statement_from_source` with the dummy result.
  - Returns the created `statement.id` (semantic layer).
- `main()`
  - Loads MySQL config using `load_config_from_env`.
  - Runs `run_example_for_first_chunk`.
  - Prints the inserted semantic statement id.

8.2 Anticipated attribution engine integration
----------------------------------------------

When the real attribution engine from `12_SPEC__Semantic_Engine_v0_4.txt`
is available, the example can be updated as follows:

- Replace `_dummy_attribution` with a call to the real engine:
  - Input: `raw_text` from `statements.text` (or directly from `chunks.text`).
  - Output: full `StatementAttribution` structure:
    - `area_id`, `family_id`.
    - `F_statement`, `A_statement`.
    - `inwardVector[12]`, `outwardVector[12]`.
    - Embedding `E_statement` for the vector DB.
- Serialize vectors:
  - Convert the relevant arrays to a compact binary or JSON encoding,
    then pass them as `bytes` into:
    - `inward_vector`, `outward_vector`, `f_vector`, `a_vector`.
- Coordinate computation:
  - Use GeometryService and the indexing spec (`13_SPEC`) to compute:
    - `side_id`, `hex_index`, `side_index` from `(area_id, family_id, ...)`.
  - Populate `SideCoord` with these canonical values instead of the
    current dummy hash.
- Domain policy:
  - Optionally use `domain_profile` and `dictionary_word` to bias or
    constrain attribution before committing it to SQL.

The surrounding DB flow (insert into `statement`, update
`statements.semantic_statement_id`, upsert `side`) remains unchanged.
